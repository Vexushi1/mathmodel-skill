solver_justification:
  governance_level: default
  first_use_chain:
    - current_mathematical_structure_or_difficulty
    - reason_solver_family_fits_or_direct_method_is_insufficient
    - solver_role_in_current_question
  precondition_chain:
    - solver_required_property
    - current_model_evidence_for_property
    - local_or_global_scope
    - solver_invocation
  precondition_profiles:
    root_or_boundary_search:
      typical_requirements: [bracket_or_candidate_interval, sign_change_monotonicity_or_local_event_structure_when_required]
    gradient_based:
      typical_requirements: [usable_local_variation, differentiability_or_valid_subgradient_when_required, feasible_update_structure]
    enumeration:
      typical_requirements: [candidate_set_completeness_or_declared_scope]
    decomposition:
      typical_requirements: [subproblem_definition, mapping_to_original_model, coupling_recovery_or_residual_scope]
    local_search:
      typical_requirements: [initialization_scope, neighborhood_or_step_definition, local_claim_scope]
    heuristic_or_population_search:
      typical_requirements: [search_domain, feasibility_handling, objective_evaluation, termination, claim_scope]
  precondition_rule: >-
    不要求在正文复述算法教科书全部理论条件，只闭合当前求解真正依赖的关键性质；若该性质仅局部成立，
    solver 的适用范围和最终 claim 也只能局部成立。
  repeated_use_rule: >-
    同一算法在后问沿用时不重复算法百科；说明继承了哪些结构，因此该算法仍适用，并只写新增编码、约束或搜索变化。
  changed_solver_rule: >-
    更换求解器时说明新增的离散性、非光滑、规模、不确定性、分解结构或可证明性质怎样改变了求解需求；
    能够恢复“为什么前问 solver 不再足够”或“为什么仍然足够”。
  alternative_method_rule: >-
    正文写“同时采用/另用某算法”时，该方法必须实际运行且有 artifact；必须标明 baseline、alternative 或 validator 角色，
    给出对应指标、误差、结构结论或未发现更优的范围，禁止为增加工作量观感而堆算法名。
  failure_rule: 仅以“算法先进、收敛快、应用广泛”作为选择理由不合格，必须回到本题模型结构。
  machine_audit_scope:
    may_review:
      - first_use_solver_without_declared_local_reason
      - solver_precondition_missing_for_declared_dependency
      - alternative_algorithm_without_declared_role_or_evidence_anchor
    must_not_claim:
      - solver_is_best_from_justification_text
      - convergence_or_global_optimality_from_algorithm_name
      - precondition_satisfaction_from_method_name_or_keywords_only

    paragraph_necessity: paragraph_necessity
    claim_strength: claim_strength_calibration
  stage_separation:
    rule: >-
      进入“问题X模型建立与求解”后默认不重新完整复述问题分析、模型假设或题目要求；
      只保留推动当前数学关系所需的短承接，以及某个既有假设对当前公式或边界的直接后果。
    review_risks:
      - repeated_problem_analysis_in_model_section
      - repeated_assumption_list_in_model_section
      - repeated_prompt_restatement_before_model_derivation
  continuous_mathematical_narrative:
    preferred_progression:
      - inherit_current_object_or_relation
      - identify_next_mathematical_need
      - introduce_only_needed_object_or_quantity
      - derive_define_or_transform_relation
      - state_structural_or_problem_consequence
      - continue_to_next_relation_final_model_or_solution
    reasoning_unit_rule: >-
      一个自然段可以承载多个彼此依赖的推理动作，只要它们围绕同一局部数学任务连续推进；
      当模型选择、solver 介绍、指标定义、结果解释等内容彼此没有直接依赖或需要不同证据来源时，应分成独立推理单元，
      不能为了“少段落”压成一个长句或一个信息过载段。
    paragraph_split_rule: >-
      分段依据语义依赖、对象切换、证据角色和下游任务变化，不按固定字符数、句数、公式数或版面长度切分。
      同一推理链很长时可以跨多个自然段连续展开；短段若已完整闭合一个独立推理单元也不因字数少而合并。
    rules:
      - 相邻公式服务同一数学任务时优先连续推演，不为每个关系机械命名一个“模型”。
      - 一个自然段可同时承担解释、公式与下游用途，只要逻辑连续；不要求“公式一段、解释一段”。
      - “建立A模型—建立B模型—采用C算法”的报告式罗列应重写为对象关系怎样逐步产生下一数学任务。
      - 连续性不能成为删除边界条件、异常、反例或必要定义的理由。
      - 连续叙事不是“越少标题越好”；真实独立的证明、结构化简、参数证据或 solver stage 若合并后降低导航性，应保留短标题。
  formula_prose_rhythm:
    consumes: formula_reasoning_chain
    functional_sequence: [need, basis, formula, meaning, consequence]
    context_recoverability_rule: >-
      对非显然公式或连续公式组，邻近正文应让读者恢复“当前已知什么、还缺什么关系、为什么引入本式、依据怎样推出、得到后有什么作用”。
      这是信息关系而不是固定句序；多个功能可合并表达，但不能省掉决定理解的桥接。
    compression_rule: >-
      need、basis、meaning、consequence 可以合并成一到数句，不要求五个标签逐项显式出现；
      对核心关系，邻近正文至少应让读者恢复“为什么此时需要该式”和“该式得到后下一步怎样变化”。
    condition_locality_rule: >-
      关键成立条件、近似范围、定义域、边界、局部性或失效条件应尽量贴近对应公式/变换首次使用处；
      若这些条件已经在邻近段或明确锚点完整给出，可引用而不重复。不得把决定公式适用性的条件统一拖到章节末尾。
    meaning_priority: >-
      符号首次出现时先完成必要定义；符号已经定义后，公式后的正文优先解释结构作用、问题转化、可行域变化、
      判据变化或下一计算用途，而不是逐个翻译符号。
    low_value_patterns:
      - source_free_related_theory_statement
      - repeated_symbol_translation_without_new_role
      - formula_declared_as_model_without_downstream_consequence
  transition_function_governance:
    principle: 不建立连接词词库；过渡句的价值由逻辑功能决定，而不是由“因此、进一步、下面”等表面词决定。

      本节主要研究”一类管理型元话语若没有同时传递对象、困难、转化、求解入口或结果含义，应删除或并入有效句。
